# 客户合作管理：改接客户入驻申请数据源 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将「客户合作管理」列表页与详情页的数据源从旧 `recruit_applications` 切换为客户入驻申请（OnboardingStore），按新字段规范重写两页，并彻底清理旧数据链路。

**Architecture:** 纯前端静态站点（HTML 页内联脚本 + 共享 `js/*.js` 模块，localStorage 存储）。数据层改动集中在 `js/onboarding-store.js`（新增对接商务字段与更新方法、入驻类型推导函数）；两个页面直接消费 Store；旧模型专属页面与死代码直接删除。测试用 `node:test`，页面测试采用「读 HTML + 正则断言 + 内联脚本编译检查」的既有模式。

**Tech Stack:** 原生 HTML/CSS/JS（ES5 风格，页内联脚本）、localStorage、node:test + node:assert/strict。

**Spec:** `docs/superpowers/specs/2026-09-04-客户合作管理-入驻申请数据源-design.md`

## Global Constraints

- 列表列顺序固定：**序号 | 客户名称 | 入驻类型 | 入驻模式 | 入驻平台 | 对接商务 | 创建时间 | 创建人 | 更新人 | 更新时间 | 状态 | 操作**
- 列表数据范围：仅 `status ∈ {submitted, reviewed}`，按 `updatedAt` 倒序
- 入驻类型映射：`DP→'DP'`；`客户KOC/联营KOC/素人KOC→'KOC'`；`达播达人/达播机构→'达播'`；其他→`'-'`
- 客户名称 = `form.finance.fullName`；创建人 = `ownerName || ownerPhone`；更新人 = 最后一条版本记录 `operator`（无则回退创建人）
- 对接商务字段名 `bdContact`，顶层字段，输入上限 50 字；已审核（reviewed）禁止修改；修改动作的版本记录文案为「更新对接商务」
- 所有空值展示为 `-`
- 页面脚本保持 ES5 风格（`var`、`function`，不用箭头函数/模板字符串），与现有代码一致
- 提交粒度：每个任务一次提交；`index.html`、`manage.html`、`panorama-customer.html` 工作区已有他人未提交改动，提交这些文件时会一并带入（已知且可接受），不要提交本任务未触及的其他已修改文件（如 `css/style.css`、`js/auth.js`、store-engine 其他页面）
- 测试基线：`node --test tests/`（项目根目录执行）

```js
  // 更新对接商务（已审核前均可修改；值未变化不落版本记录）
  function updateBdContact(id, bdContact, operator) {
    var list = readAll();
    for (var i = 0; i < list.length; i++) {
      if (list[i].id !== id) continue;
      var app = list[i];
      if (app.status === STATUS.REVIEWED) {
        return { ok: false, error: '当前状态（' + statusLabel(app.status) + '）不允许修改对接商务' };
      }
      var newValue = String(bdContact || '').trim();
      var oldValue = app.bdContact || '';
      if (newValue === oldValue) return { ok: true, app: clone(app) };
      app.bdContact = newValue;
      app.updatedAt = nowText();
      pushVersion(app, '更新对接商务', ['对接商务：' + (oldValue || '-') + ' → ' + (newValue || '-')], operator);
      writeAll(list);
      return { ok: true, app: clone(app) };
    }
    return { ok: false, error: '申请不存在' };
  }
```

function openBdContactModal() {
  if (!app) return;

  document.getElementById('bdContactModal').hidden = false;
  document.getElementById('bdContactInput').focus();

function closeBdContactModal() {
  document.getElementById('bdContactModal').hidden = true;
}

function confirmBdContact() {
  if (!app) return;
  var value = document.getElementById('bdContactInput').value.trim();
  var user = Auth.getCurrentUser() || {};
  var result = Store.updateBdContact(app.id, value, user.name || user.phone || '运营');
  if (!result.ok) { showToast(result.error, 'error'); return; }
  app = result.app;
  closeBdContactModal();
  renderHead();
  showToast('对接商务已更新', 'success');
}

// 四步表单只读展示（按 Store.STEPS / FIELD_DEFS 渲染）
function renderFormCards() {
  var container = document.getElementById('dFormCards');
  var html = '';
  Store.STEPS.forEach(function(step) {
    html += '<div class="detail-card"><h3 class="detail-card-title locked">' + esc(step.label) + '</h3><div class="detail-grid">';
    Store.FIELD_DEFS[step.key].forEach(function(def) {
      if (def.showIf && app.form[step.key][def.showIf.key] !== def.showIf.value) return;
      var value = app.form[step.key][def.key];
      if (def.type === 'radio' && def.options) {
        var opt = def.options.filter(function(o) { return o.value === value; })[0];
        value = opt ? opt.label : value;
      }
      html += '<div class="detail-item"><div class="k">' + esc(def.label) + (def.unit ? '（' + def.unit + '）' : '') + '</div><div class="v">' + esc(value || '-') + '</div></div>';
    });
    if (step.key === 'plan') html += renderMonthly();
    html += '</div></div>';
  });
  container.innerHTML = html;
}

// 月度分解目标（1-12 月）表格
function renderMonthly() {
  var monthly = (app.form.plan && app.form.plan.monthly) || {};
  var cells = '';
  var sum = 0;
  Store.MONTH_KEYS.forEach(function(mk) {
    var v = Number(monthly[mk] || 0);
    sum += v;
    cells += '<td>' + v.toLocaleString('zh-CN') + '</td>';
  });
  var head = Store.MONTH_KEYS.map(function(mk, i) { return '<th>' + (i + 1) + '月</th>'; }).join('');
  return '<div style="grid-column:1/-1;">' +
    '<div class="k" style="font-size:12px;color:var(--text-tertiary,#94a3b8);margin-bottom:8px;">月度分解目标（万元）</div>' +
    '<div style="overflow-x:auto;"><table class="month-table"><thead><tr>' + head + '<th>合计</th></tr></thead>' +
    '<tbody><tr>' + cells + '<td style="background:#fffbeb;font-weight:600;">' + sum.toLocaleString('zh-CN') + '</td></tr></tbody></table></div></div>';
}

function renderVersions() {
  var versions = app.versions || [];
  var body = document.getElementById('dVersionBody');
  if (!versions.length) {
    body.innerHTML = '<tr><td colspan="5" style="text-align:center;color:#94a3b8;">暂无版本记录</td></tr>';
    return;
  }
  body.innerHTML = versions.map(function(v) {
    return '<tr><td>v' + v.v + '</td><td>' + esc(v.time) + '</td><td>' + esc(v.operator) + '</td><td>' + esc(v.action) + '</td><td>' + esc((v.changes || []).join('；') || '-') + '</td></tr>';
  }).join('');
}

render();
